Структура плагина

Единый .js-файл. Мета — в комментариях в самом начале.

Core
// @name Название Плагина
// @author Ваш Ник
// @version 1.0
// @description Описание плагина.
// @banner https://funpay.tools/default-banner.jpeg

// PLUGIN_ID и PLUGIN_SLOT_KEY доступны глобально — ядро их инжектирует.
fpt.app.log("Плагин запущен. ID: " + PLUGIN_ID);
Запрещено Нельзя объявлять PLUGIN_ID, PLUGIN_SLOT_KEY, вызывать fpt.plugin.register(), fpt.init(), использовать fpt.config.*, писать while(true).

Kotlin Bridge — что добавить в проект

Весь этот блок добавляется в класс FPTJsBridge в файле Plugins.kt, а JS-часть — в константу API_BOOTSTRAP_JS.

Код для реализации
Как добавить 1. В FPTJsBridge добавь все @JavascriptInterface fun ... методы.
2. В API_BOOTSTRAP_JS в блок window.fpt = { ... } добавь новые объекты модулей.
3. Пересобери проект.

Canvas → Image

Генерация и отправка картинок через WebView Canvas.

Строгий порядок
Обязательная цепочка fetchImageBase64(url) → установить img.src → нарисовать на Canvas → canvas.toDataURL() → saveBase64Image() → sendWithImage() 
var b64 = fpt.network.fetchImageBase64("https://example.com/img.png");
var img = new Image();
img.onload = function() {
    var c = document.createElement("canvas");
    c.width = 400; c.height = 200;
    var ctx = c.getContext("2d");
    ctx.fillStyle = "#1a1a2e"; ctx.fillRect(0, 0, 400, 200);
    ctx.drawImage(img, 10, 50, 100, 100);
    ctx.fillStyle = "#fff"; ctx.font = "bold 24px sans-serif";
    ctx.fillText("Привет!", 130, 110);
    var uri = fpt.app.saveBase64Image(c.toDataURL("image/png"));
    if (uri) fpt.chat.sendWithImage("chat_id_here", "Картинка", uri, true);
};
img.src = b64;
